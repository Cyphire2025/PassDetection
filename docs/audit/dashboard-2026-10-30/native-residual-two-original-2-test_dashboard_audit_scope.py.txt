def policy(report, **changes):
    document = {
        "schema_version": 1,
        "deferred_patterns": ["backend/app/infrastructure/email/*", "backend/tests/**/test_email_*.py"],
        "retained_overrides": ["backend/tests/unit/infrastructure/test_email_storage_reconciliation.py"],
        **changes,
    }
    (report / "audit-scope.json").write_text(json.dumps(document), encoding="utf-8")

def test_active_totals_exclude_deferred_files_but_keep_original_review_status(tmp_path, monkeypatch):
    report = tmp_path / "audit"
    report.mkdir()
    inbox = "backend/app/infrastructure/email/tasks.py"
    retained = "backend/app/passports.py"
    for name in (inbox, retained):
        path = tmp_path / name
        path.parent.mkdir(parents=True, exist_ok=True)
        path.write_text("pass\n", encoding="utf-8")
    policy(report)
    record = {"path": inbox, "sha256": hashlib.sha256((tmp_path / inbox).read_bytes()).hexdigest(),
              "review_scope": "full", "status": "needs-follow-up", "notes": "Independent finding remains open."}
    ledger = report / "context-reviewed-files.json"
    original = json.dumps([record]).encode()
    ledger.write_bytes(original)
    monkeypatch.setattr(inventory, "ROOT", tmp_path)
    monkeypatch.setattr(inventory, "REPORT", report)
    monkeypatch.setattr(inventory.subprocess, "check_output", lambda *_args, **_kwargs: (inbox + "\0" + retained + "\0").encode())
    summary = inventory.refresh([], "reviewed", "test", "")
    with (report / "file-inventory.csv").open(encoding="utf-8", newline="") as stream:
        rows = {row["path"]: row for row in csv.DictReader(stream)}
    assert rows[inbox]["scope"] == "operations-inbox-deferred"
    assert rows[inbox]["status"] == "needs-follow-up" and rows[inbox]["review_scope"] == "full"
    assert rows[inbox]["sha256"] == record["sha256"] and ledger.read_bytes() == original
    assert summary["active_files"] == 1 and summary["active_review_scope"] == {"unreviewed": 1}
    assert summary["deferred_by_user"]["files"] == 1
    assert summary["deferred_by_user"]["review_scope"] == {"full": 1}
    assert all(row["path"] != inbox for row in summary["largest_source_modules"])

def test_only_four_explicit_public_sample_assets_can_enter_retirement_history(tmp_path, monkeypatch):
    names = ("passport-front-cover.png", "passport-back-cover.png", "visa-photo.png", "README.md")
    records = [{"path": "frontend/public/assets/upload-samples/" + name, "sha256": "a" * 64}
               for name in names]
    manifest = tmp_path / "retired-upload-samples-manifest.json"
    original = json.dumps({"files": records}).encode()
    manifest.write_bytes(original)
    monkeypatch.setattr(inventory, "REPORT", tmp_path)
    history = inventory.retired_source_history()
    assert set(history) == {row["path"] for row in records}
    assert all(row["review_scope"] == "retired" and row["sha256"] == "a" * 64 for row in history.values())
    assert manifest.read_bytes() == original
    records.append({"path": "frontend/public/unrelated.png", "sha256": "b" * 64})
    manifest.write_text(json.dumps({"files": records}), encoding="utf-8")
    with pytest.raises(ValueError):
        inventory.retired_source_history()


def foundation_manifest():
    paths = (
        "backend/app/application/dtos/whatsapp_dtos.py",
        "backend/app/application/interfaces/whatsapp_provider.py",
        "backend/app/application/use_cases/whatsapp/plan_group_broadcast_use_case.py",
        "backend/app/infrastructure/whatsapp/noop_provider.py",
        "backend/app/infrastructure/whatsapp/pricing.py",
    )
    return {
        "schema_version": 1,
        "files": [
            {
                "path": path,
                "original_path": path,
                "archive_path": "Not related to project/retired-whatsapp-foundation/" + path,
                "sha256": hashlib.sha256(path.encode()).hexdigest(),
                "archive_sha256": hashlib.sha256(path.encode()).hexdigest(),
                "physical_lines": index + 1,
                "bytes": 100 + index,
                "status": "retired",
            }
            for index, path in enumerate(paths)
        ],
    }


def test_exact_five_foundation_sources_remain_visible_without_local_archive(tmp_path, monkeypatch):
    report = tmp_path / "audit"
    report.mkdir()
    document = foundation_manifest()
    manifest = report / "whatsapp-foundation-retirement-manifest.json"
    original = json.dumps(document).encode()
    manifest.write_bytes(original)
    monkeypatch.setattr(inventory, "REPORT", report)
    monkeypatch.setattr(inventory, "ROOT", tmp_path)
    monkeypatch.setattr(inventory.subprocess, "check_output", lambda *_args, **_kwargs: b"")
    summary = inventory.refresh([], "reviewed", "test", "")
    with (report / "file-inventory.csv").open(encoding="utf-8", newline="") as stream:
        rows = {row["path"]: row for row in csv.DictReader(stream)}
    assert set(rows) == {record["path"] for record in document["files"]}
    for record in document["files"]:
        row = rows[record["path"]]
        assert row["sha256"] == record["sha256"]
        assert int(row["lines"]) == record["physical_lines"]
        assert row["scope"] == "backend" and row["review_scope"] == "retired"
        assert row["status"] == "removed-in-working-tree"
    assert summary["active_files"] == 0 and summary["files"] == 5
    assert manifest.read_bytes() == original


@pytest.mark.parametrize("change", [
    "backend-path", "frontend-path", "traversal", "archive-path", "archive-hash",
    "original-path", "schema", "duplicate", "incomplete", "byte-count",
])
def test_foundation_retirement_cannot_admit_unrelated_or_inconsistent_records(
    tmp_path, monkeypatch, change,
):
    document = foundation_manifest()
    record = document["files"][0]
    if change == "backend-path":
        record["path"] = "backend/app/infrastructure/whatsapp/template_settings.py"
    elif change == "frontend-path":
        record["path"] = "frontend/features/whatsapp/live.ts"
    elif change == "traversal":
        record["path"] = "backend/app/application/dtos/../../outside.py"
    elif change == "archive-path":
        record["archive_path"] = "Not related to project/../../outside.py"
    elif change == "archive-hash":
        record["archive_sha256"] = "b" * 64
    elif change == "original-path":
        record["original_path"] = "backend/unrelated.py"
    elif change == "schema":
        document["schema_version"] = True
    elif change == "duplicate":
        document["files"].append(dict(record))
    elif change == "incomplete":
        document["files"].pop()
    elif change == "byte-count":
        record["bytes"] = True
    (tmp_path / "whatsapp-foundation-retirement-manifest.json").write_text(
        json.dumps(document), encoding="utf-8",
    )
    monkeypatch.setattr(inventory, "REPORT", tmp_path)
    with pytest.raises(ValueError):
        inventory.retired_source_history()
