"""Record every actual outcome of a predeclared bounded repeated collection."""
from __future__ import annotations

import json
import os
from pathlib import Path

EXPECTED = "tests/unit/infrastructure/test_traveller_reconciliation_runtime.py::test_burst_deadline_also_cancels_blocked_admission"
OUTPUT = Path(os.environ["AUDIT_DEADLINE_PROBE_LOG"])
ORDINAL = 0


def pytest_collection_modifyitems(items):
    assert len(items) == 50, f"Expected exactly50 actual collected invocations, got{len(items)}"
    assert all(item.nodeid == EXPECTED and item.originalname == EXPECTED.split("::")[1] for item in items)


def pytest_runtest_logstart(nodeid, location):
    global ORDINAL
    assert nodeid == EXPECTED
    ORDINAL += 1
    assert ORDINAL <= 50


def pytest_runtest_logreport(report):
    record = {"invocation": ORDINAL, "nodeid": report.nodeid, "when": report.when,
              "outcome": report.outcome, "seconds": report.duration,
              "longrepr": report.longreprtext if report.failed else None}
    with OUTPUT.open("a", encoding="utf-8", newline="\n") as stream:
        stream.write(json.dumps(record) + "\n")
